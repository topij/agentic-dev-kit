def git(root, *args):
    return subprocess.check_output(
        ['git', '--no-optional-locks', '-C', str(root), *args])


def digest(data):
    return hashlib.sha256(data).hexdigest()


def inventory(root):
    result = {}
    for path in sorted(root.rglob('*')):
        rel = path.relative_to(root).as_posix()
        if rel == '.git' or rel.startswith('.git/'):
            continue
        entry = {'mode': oct(path.lstat().st_mode & 0o777)}
        if path.is_symlink():
            entry.update(kind='symlink', target=os.readlink(path))
        elif path.is_file():
            entry.update(kind='file', sha256=digest(path.read_bytes()))
        elif path.is_dir():
            entry.update(kind='directory')
        else:
            raise ValueError(path)
        result[rel] = entry
    return result


def git_inventory(ref):
    result = {}
    for row in git(COCKPIT, 'ls-tree', '-rz', ref).split(b'\0'):
        if not row:
            continue
        metadata, path = row.split(b'\t')
        mode, kind, oid = metadata.split()
        assert kind == b'blob', (ref, path, kind)
        data = git(COCKPIT, 'cat-file', 'blob', oid.decode())
        result[path.decode()] = {'git_mode': mode.decode(), 'blob': oid.decode(),
                                 'sha256': digest(data)}
    return result
