def canon(x):
    # This fixture uses no floating point numbers. Reject unsupported numbers,
    # rather than claim Python float rendering is ECMAScript serialization.
    def check(v):
        if v is None or isinstance(v,(str,bool)): return
        if type(v) is int and abs(v)<=9007199254740991:return
        if isinstance(v,list):
            for y in v:check(y)
            return
        if isinstance(v,dict):
            for k,y in v.items():
                if not isinstance(k,str) or not k.isascii():raise ValueError('canonicalizer requires ASCII mapping keys')
                check(y)
            return
        raise ValueError('unsupported canonical scalar: '+repr(v))
    check(x)
    return json.dumps(x,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode('utf-8')
def digest(x):return 'sha256:'+hashlib.sha256(canon(x)).hexdigest()
