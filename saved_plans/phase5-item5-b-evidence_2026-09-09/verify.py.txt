def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_json(name, value):
    assert Path.cwd().resolve() in (REPO, KIT)
    with (OUT / name).open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')


def run(name, argv, cwd, check=True, extra_env=None):
    os.chdir(cwd)
    assert Path.cwd().resolve() == cwd
    stamp = {'argv': argv, 'cwd': str(cwd), 'source_pin': authority['source_pin'],
             'revision': subprocess.check_output(['git', '-C', str(cwd), 'rev-parse', 'HEAD'], text=True).strip(),
             'status_before': subprocess.check_output(['git', '-C', str(cwd), 'status', '--short'], text=True),
             'started_at': datetime.now(timezone.utc).isoformat(), 'environment_overrides': extra_env or {}}
    with (OUT / (name + '.log')).open('x') as stream:
        result = subprocess.run(argv, cwd=cwd, env={**os.environ, **(extra_env or {})}, stdout=stream, stderr=subprocess.STDOUT)
    stamp['exit_code'] = result.returncode
    stamp['finished_at'] = datetime.now(timezone.utc).isoformat()
    stamp['log_sha256'] = sha(OUT / (name + '.log'))
    stamp['status_after'] = subprocess.check_output(['git', '-C', str(cwd), 'status', '--short'], text=True)
    write_json(name + '.json', stamp)
    print(name, json.dumps(stamp), flush=True)
    print('\n'.join((OUT / (name + '.log')).read_text().splitlines()[-25:]), flush=True)
    if check:
        result.check_returncode()
    return result.returncode


def inventory(root):
    result = {}
    for path in sorted(root.rglob('*')):
        rel = path.relative_to(root).as_posix()
        if rel == '.git' or rel.startswith('.git/'):
            continue
        st = path.lstat()
        entry = {'mode': oct(st.st_mode & 0o777)}
        if path.is_symlink():
            entry.update(kind='symlink', target=os.readlink(path))
        elif path.is_file():
            entry.update(kind='file', sha256=sha(path))
        elif path.is_dir():
            entry.update(kind='directory')
        else:
            raise AssertionError(str(path))
        result[rel] = entry
    return result
