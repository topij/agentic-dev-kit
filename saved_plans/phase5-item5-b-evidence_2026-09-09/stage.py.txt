"""Execute only the approved ITEM5-B local staging and record destination evidence."""
import copy
import hashlib
import json
import os
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

COCKPIT = Path('/Users/topi/Coding/agentic-dev-kit')
PARENT = Path('/Users/topi/Coding/adk-field-exercises/item5-b-20260909')
REPO = PARENT / 'fixture'
KIT = PARENT / 'kit-source'
OUT = PARENT / 'evidence'
PIN = '8418118e40728c667c32a28a182697139bc7a5ef'
SAVED = COCKPIT / 'saved_plans'
os.environ['PYTHONDONTWRITEBYTECODE'] = '1'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_json(name, value):
    assert Path.cwd() in (COCKPIT, REPO, KIT)
    with (OUT / name).open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')


def run(name, argv, cwd):
    os.chdir(cwd)
    assert Path.cwd().resolve() == cwd
    result = subprocess.run(argv, cwd=cwd, text=True, capture_output=True)
    record = dict(argv=argv, cwd=str(cwd), observed_at=datetime.now(timezone.utc).isoformat(),
                  exit_code=result.returncode, stdout=result.stdout, stderr=result.stderr)
    write_json(name + '.json', record)
    print(name, result.returncode, result.stdout[-1800:], result.stderr[-1800:], flush=True)
    result.check_returncode()
    return result.stdout


assert sys.argv[1] == 'stage'
assert Path.cwd().resolve() == COCKPIT
assert subprocess.check_output(['git', '-C', str(COCKPIT), 'rev-parse', 'HEAD'], text=True).strip() == 'edc199f42bc65b1850172ea79b791ef358577ae1'
for original in [Path('/private/tmp/adk-adopt-field-20260905-5mfj1st8/fixture'),

run('clone', ['git', 'clone', '--no-hardlinks', '--no-checkout', str(COCKPIT), str(KIT)], COCKPIT)

    source, target = KIT / name, REPO / dest
