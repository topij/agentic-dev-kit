"""Validate the hash-bound UPDATE-03 proposal and rerun its read-only audit."""
import gzip
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
from datetime import datetime, timezone

COCKPIT = Path('/Users/topi/Coding/agentic-dev-kit')
SAVED = COCKPIT / 'saved_plans'
EVIDENCE = SAVED / 'phase5-item5-b-update03-evidence_2026-09-11'
AUDIT = SAVED / 'phase5-item5-b-update03-audit-r4_2026-09-11.py.txt'
BINDING = EVIDENCE / 'prepared-input-binding-r5.json'


def sha(data):
    return hashlib.sha256(data).hexdigest()


def main():
    if sys.flags.optimize:
        raise SystemExit('Refusing optimized Python: validation assertions must execute')
    assert sys.dont_write_bytecode and Path.cwd() == COCKPIT
    assert len(sys.argv) == 3 and sys.argv[1] == '--binding-sha256'
    binding_bytes = BINDING.read_bytes()
    assert sha(binding_bytes) == sys.argv[2], 'binding differs from the exact decision'
    binding = json.loads(binding_bytes)
    assert binding['decision'] == 'ITEM5-B-UPDATE-03' and binding['approved'] is False
    files = binding['files']
    expected = {str(p.relative_to(COCKPIT)) for p in [
        AUDIT, Path(__file__).resolve(),
        SAVED / 'phase5-item5-b-update03-command-r4_2026-09-11.py.txt',
        SAVED / 'phase5-item5-b-acceptance-audit_2026-09-10.py.txt',
        SAVED / 'phase5-item5-b-review-followup-evidence_2026-09-10/retained-final.json.gz',
        SAVED / 'phase5-item5-b-acceptance-execution-evidence_2026-09-10/forge-after-exit.json',
        SAVED / 'phase5-item5-b-update-evidence_2026-09-10/configuration-final.json',
        SAVED / 'phase5-item5-b-update-evidence_2026-09-10/replay-evidence-final.json',
        EVIDENCE / 'audit-r4.json.gz', EVIDENCE / 'proposed-writes-r4.json',
        EVIDENCE / 'forge-readback.json.gz', EVIDENCE / 'source-delivery.json.gz',
        EVIDENCE / 'source-followup-findings.json.gz',
        EVIDENCE / 'payloads/kit-manifest.json.txt',
    ]}
    payload_keys = {'scripts/conftest.py', 'scripts/tests/test_state_guard.py',
                    'scripts/tests/test_init_sh.py', 'scripts/tests/test_portability.py',
                    'docs/agentic-dev-kit/workflows/upgrade.md'}
    expected.update(str((EVIDENCE / 'payloads' / (key + '.txt')).relative_to(COCKPIT))
                    for key in payload_keys)
    assert set(files) == expected, 'unexpected binding scope'
    for name, digest in files.items():
        path = COCKPIT / name
        assert path.resolve() == path and path.is_file(), name
        assert sha(path.read_bytes()) == digest, ('bound file changed', name)
    recorded = json.loads(gzip.decompress((EVIDENCE / 'audit-r4.json.gz').read_bytes()))
    program_digest = files[str(AUDIT.relative_to(COCKPIT))]
    assert recorded['program_sha256'] == program_digest, 'recorded audit/program mismatch'
    assert recorded['helper_sha256'] == files['saved_plans/phase5-item5-b-acceptance-audit_2026-09-10.py.txt']
    assert recorded['checkpoint_sha256'] == files['saved_plans/phase5-item5-b-review-followup-evidence_2026-09-10/retained-final.json.gz']
    assert recorded['approved'] is False and recorded['decision'] == binding['decision']
    ledger = json.loads((EVIDENCE / 'proposed-writes-r4.json').read_bytes())
    assert ledger['audit_sha256'] == files[str((EVIDENCE / 'audit-r4.json.gz').relative_to(COCKPIT))]
    assert {row['source_key'] for row in ledger['fixture_payload_writes']} == payload_keys
    for row in ledger['fixture_payload_writes']:
        payload = EVIDENCE / 'payloads' / (row['source_key'] + '.txt')
        assert sha(payload.read_bytes()) == row['after']['sha256'], ('payload differs', payload)
    assert sha((EVIDENCE / 'payloads/kit-manifest.json.txt').read_bytes()) == ledger['baseline_write']['after_sha256']
    env = dict(os.environ)
    env.pop('PYTHONOPTIMIZE', None)
    argv = [sys.executable, '-B', str(AUDIT)]
    run = subprocess.run(argv, cwd=COCKPIT, env=env, capture_output=True, text=True, check=True)
    fresh = json.loads(run.stdout)
    assert fresh['program_sha256'] == program_digest, 'fresh audit/program mismatch'
    for key in ledger:
        if key != 'audit_sha256':
            assert recorded[key] == fresh[key] == ledger[key], ('proposal changed', key)
    print(json.dumps(dict(approved=False, decision=binding['decision'],
                         result='prepared-inputs-and-program-binding-verified; proposal-only',
                         argv=sys.orig_argv, cwd=str(COCKPIT),
                         revision=fresh['revision'],
                         observed_at=datetime.now(timezone.utc).isoformat(),
                         binding_sha256=sha(binding_bytes), audit_argv=argv,
                         audit_stderr=run.stderr, fresh_audit=fresh), indent=2, sort_keys=True))


if __name__ == '__main__':
    main()
