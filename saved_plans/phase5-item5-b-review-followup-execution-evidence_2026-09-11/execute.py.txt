def now(): return datetime.now(timezone.utc).isoformat()
def sha(b): return hashlib.sha256(b).hexdigest()
def read(p): return json.loads(p.read_bytes())
def git(p, *args):
    return subprocess.check_output(['git','--no-optional-locks','-C',str(p),*args],text=True).strip()
def enter(p):
    assert p.resolve() == p and p.is_dir() and not p.is_symlink()
    os.chdir(p)
    assert Path.cwd() == p
def write(p, obj):
    assert p.is_absolute() and p.resolve() == p
    if p.exists(): assert p.is_file() and p.stat().st_nlink == 1
    p.write_text(json.dumps(obj,indent=2,sort_keys=True)+'\n')
def env(role='author'):
    enter(OUT)
    owned=OUT/role; owned.mkdir(exist_ok=True)
    for name in ['state','uv-cache','uv-tools','ruff-cache','pytest-cache','tmp']:
        (owned/name).mkdir(exist_ok=True)
    e=dict(os.environ); e.pop('PYTHONOPTIMIZE',None)
    e.update(PYTHONDONTWRITEBYTECODE='1', DEVKIT_STATE_ROOT=str(owned/'state'), UV_CACHE_DIR=str(owned/'uv-cache'), UV_TOOL_DIR=str(owned/'uv-tools'), RUFF_CACHE_DIR=str(owned/'ruff-cache'), TMPDIR=str(owned/'tmp'), PYTEST_ADDOPTS=f'-o cache_dir={owned/"pytest-cache"} -ra')
    return e
def call(name, argv, cwd=KIT, accepted=(0,), role='author'):
    e=env(role); enter(cwd)
    started=now(); revision=git(cwd,'rev-parse','HEAD')
    print(json.dumps(dict(starting=name,argv=argv,cwd=str(cwd),revision=revision,at=started)),flush=True)
    with (OUT/(name+'.stdout.log')).open('xb') as out, (OUT/(name+'.stderr.log')).open('xb') as err:
        p=subprocess.run(argv,cwd=cwd,env=e,stdout=out,stderr=err)
    obj=dict(argv=argv,cwd=str(cwd),revision=revision,started_at=started,ended_at=now(),returncode=p.returncode,environment={k:e[k] for k in ['PYTHONDONTWRITEBYTECODE','DEVKIT_STATE_ROOT','UV_CACHE_DIR','UV_TOOL_DIR','RUFF_CACHE_DIR','TMPDIR','PYTEST_ADDOPTS']},PYTHONOPTIMIZE=e.get('PYTHONOPTIMIZE'),stdout_sha256=sha((OUT/(name+'.stdout.log')).read_bytes()),stderr_sha256=sha((OUT/(name+'.stderr.log')).read_bytes()))
    write(OUT/(name+'.json'),obj)
    print(json.dumps(obj),flush=True)
    print((OUT/(name+'.stdout.log')).read_text(errors='replace')[-4500:],flush=True)
    print((OUT/(name+'.stderr.log')).read_text(errors='replace')[-1500:],flush=True)
    assert p.returncode in accepted,(name,p.returncode)
    return obj

def retained(label):
    enter(KIT)
    expected=json.loads(gzip.decompress((KIT/'saved_plans/phase5-item5-b-review-followup-evidence_2026-09-10/retained-final.json.gz').read_bytes()))
    actual=dict(observed_at=now(),cockpit_revision=git(KIT,'rev-parse','HEAD'),original_paths={p:os.path.lexists(p) for p in expected['original_paths']},trees={role:dict(path=str(p),inventory=inventory(p),identity=identity(p),administrative_inventory=inventory(p/'.git')) for role,p in [('fixture',REPO),('source',SOURCE)]})
    assert actual['trees']==expected['trees'],'Retained trees differ from post-acceptance checkpoint'
    assert actual['original_paths']==expected['original_paths'] and not any(actual['original_paths'].values())
    enter(OUT)
    write(OUT/('retained-'+label+'.json'),actual)
    print('Retained file/admin inventories match; original paths remain absent.',flush=True)
    return actual

def preflight():
    enter(KIT)
    assert git(KIT,'rev-parse','HEAD')==BASE and git(KIT,'symbolic-ref','--short','HEAD')==BRANCH and git(KIT,'status','--porcelain')==''
    assert git(KIT,'remote','get-url','origin') in ['https://github.com/topij/agentic-dev-kit.git','git@github.com:topij/agentic-dev-kit.git']
    for rel,h in read(PROPOSAL/'sha256.json').items(): assert sha((PROPOSAL/rel).read_bytes())==h,rel
    ledger=read(PROPOSAL/'write-ledger.json')
    originals={}
    for row in ledger['writes']:
        p=Path(row['destination']); s=p.lstat()
        assert p.resolve()==p and stat.S_ISREG(s.st_mode) and s.st_nlink==1
        assert sha(p.read_bytes())==row['before_sha256'] and oct(stat.S_IMODE(s.st_mode))==row['mode']
        assert sha((PROPOSAL/row['payload']).read_bytes())==row['after_sha256']
        originals[str(p)]=dict(sha256=sha(p.read_bytes()),mode=oct(stat.S_IMODE(s.st_mode)))
    assert sha((KIT/'CHANGELOG.md').read_bytes())==ledger['changelog_sha256_unchanged']
    call('kit-preflight',['gh','pr','view','731','--repo','topij/agentic-dev-kit','--json','url,state,isDraft,headRefOid,baseRefOid,headRefName,baseRefName,body,statusCheckRollup'])
    pr=read(OUT/'kit-preflight.stdout.log')
    assert pr['headRefOid']==BASE and pr['state']=='OPEN' and not pr['isDraft'] and pr['url']=='https://github.com/topij/agentic-dev-kit/pull/731' and pr['headRefName']==BRANCH
    call('protected-main',['gh','api','repos/topij/agentic-dev-kit/branches/main'])
    main=read(OUT/'protected-main.stdout.log'); assert main['protected'] and main['commit']['sha']==pr['baseRefOid']=='f408039dc3a9b0d2e0e8d246c126af892839cee8'
    for lens in ['adversarial','correctness']:
        receipt=read(KIT/'saved_plans/phase5-item5-b-review-followup-evidence_2026-09-10'/lens/'receipt.json')['forge_comment']
        call('prior-review-'+lens,['gh','api','repos/topij/agentic-dev-kit/issues/comments/'+str(receipt['response']['id'])])
        actual=read(OUT/('prior-review-'+lens+'.stdout.log'))
        assert sha(actual['body'].encode())==receipt['body_sha256']
    call('prior-checkpoint',['gh','api','repos/topij/agentic-dev-kit/issues/comments/5624755093'])
    retained('before'); fixture_forge('before')
    enter(OUT); (OUT/'before').mkdir()
    for row in ledger['writes']:
        p=Path(row['destination']); d=OUT/'before'/p.relative_to(KIT); d.parent.mkdir(parents=True,exist_ok=True); d.write_bytes(p.read_bytes()); os.chmod(d,int(row['mode'],8)); assert sha(d.read_bytes())==row['before_sha256']
    (OUT/'approved-packet.md.txt').write_bytes(PACKET.read_bytes())
    write(OUT/'before.json',originals)
    write(OUT/'authority.json',dict(decision='ITEM5-B-KIT-REVIEW-02',operator_message='I approve ITEM5-B-KIT-REVIEW-02 as scoped: accept and document the inherited special-file-root limitation for this regular-file repair, apply the hash-bound root-symlink compatibility tests and release manifest, run required verification and complete ready PR #731 through independent review, holding its reviewed head for an operator merge decision and leaving retained trees, baseline, clients, trackers and fixture PR merge untouched',recorded_at=now(),base_revision=BASE,KIT=str(KIT),REPO=str(REPO),SOURCE=str(SOURCE),packet_sha256=sha(PACKET.read_bytes()),ledger_sha256=sha((PROPOSAL/'write-ledger.json').read_bytes()),proposal_index_sha256=sha((PROPOSAL/'sha256.json').read_bytes()),special_file_root_limitation_accepted=True,merge_authorized=False,retained_writes_authorized=False))
    print('Preflight matched. Authority and original destination bytes retained.',flush=True)

def apply():
    enter(KIT)
    assert git(KIT,'rev-parse','HEAD')==BASE and git(KIT,'status','--porcelain')=='' and git(KIT,'symbolic-ref','--short','HEAD')==BRANCH
    assert read(OUT/'authority.json')['decision']=='ITEM5-B-KIT-REVIEW-02'
    rows=read(PROPOSAL/'write-ledger.json')['writes']
    for row in rows:
        p=Path(row['destination']); assert sha(p.read_bytes())==row['before_sha256'] and oct(stat.S_IMODE(p.stat().st_mode))==row['mode']
    for row in rows:
        p=Path(row['destination']); payload=(PROPOSAL/row['payload']).read_bytes(); assert sha(payload)==row['after_sha256']; p.write_bytes(payload); assert sha(p.read_bytes())==row['after_sha256']
    enter(OUT); write(OUT/'application.json',dict(at=now(),base=BASE,branch=BRANCH,destinations={r['destination']:sha(Path(r['destination']).read_bytes()) for r in rows}))
    print('Approved payload hashes verified at intended destinations.',flush=True)
