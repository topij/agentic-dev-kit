ROOT = Path(__file__).resolve().parents[1]

def digest(data):
    return hashlib.sha256(data).hexdigest()


def read(name):
    return json.loads((SAVED / name).read_text())


def git(*args):
    return subprocess.check_output(["git", "-C", str(ROOT), *args])


pool = {}


def retain(data, location):
    pool.setdefault(digest(data), location)


def string_leaves(value, location):
    if isinstance(value, str):
        retain(value.encode(), location)
    elif isinstance(value, dict):
        for key, child in value.items():
            string_leaves(child, location + "/" + str(key))
    elif isinstance(value, list):
        for index, child in enumerate(value):
            string_leaves(child, location + "/" + str(index))


source_blobs = {}
for entry in git("ls-tree", "-r", SOURCE).decode().splitlines():
    metadata, path = entry.split("\t", 1)
    mode, kind, oid = metadata.split()
    if kind == "blob":
        data = git("cat-file", "blob", oid)
        source_blobs[path] = digest(data)
        retain(data, "git:" + SOURCE + ":" + path)
for path in sorted(SAVED.rglob("*")):
    if not path.is_file() or path.name.startswith("phase5-item5-"):
        continue
    data = path.read_bytes()
    retain(data, str(path.relative_to(ROOT)))
    if path.suffix == ".json":
        string_leaves(json.loads(data), str(path.relative_to(ROOT)))
